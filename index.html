-- Crear el Bucket de Storage 'jugadores' (corrige error 'Bucket not found'):
INSERT INTO storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
VALUES ('jugadores', 'jugadores', true, 5242880, ARRAY['image/jpeg', 'image/png', 'image/webp', 'image/gif', 'image/svg+xml'])
ON CONFLICT (id) DO UPDATE SET public = true;

-- Políticas de acceso para subida y lectura pública de fotos:
DROP POLICY IF EXISTS "Lectura publica fotos jugadores" ON storage.objects;
CREATE POLICY "Lectura publica fotos jugadores" ON storage.objects FOR SELECT TO anon, authenticated USING (bucket_id = 'jugadores');

DROP POLICY IF EXISTS "Subida publica fotos jugadores" ON storage.objects;
CREATE POLICY "Subida publica fotos jugadores" ON storage.objects FOR INSERT TO anon, authenticated WITH CHECK (bucket_id = 'jugadores');

-- 1. Tabla de Jugadores (por si aún no se ha creado)
CREATE TABLE IF NOT EXISTS public.jugadores (
  dorsal INTEGER PRIMARY KEY,
  nombre TEXT NOT NULL,
  fecha_nacimiento DATE NOT NULL,
  localidad TEXT DEFAULT 'Almería',
  foto TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Si la tabla jugadores ya fue creada previamente, añade las columnas faltantes:
ALTER TABLE public.jugadores ADD COLUMN IF NOT EXISTS localidad TEXT DEFAULT 'Almería';
ALTER TABLE public.jugadores ADD COLUMN IF NOT EXISTS foto TEXT;

-- 2. Almacenamiento (Storage) para subir fotos de jugadores a Supabase
INSERT INTO storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
VALUES ('jugadores', 'jugadores', true, 5242880, ARRAY['image/jpeg', 'image/png', 'image/webp', 'image/gif'])
ON CONFLICT (id) DO UPDATE SET public = true;

CREATE POLICY "Fotos jugadores lectura" ON storage.objects FOR SELECT TO public USING (bucket_id = 'jugadores');
CREATE POLICY "Fotos jugadores subida" ON storage.objects FOR INSERT TO public WITH CHECK (bucket_id = 'jugadores');
CREATE POLICY "Fotos jugadores edicion" ON storage.objects FOR UPDATE TO public USING (bucket_id = 'jugadores');
CREATE POLICY "Fotos jugadores borrado" ON storage.objects FOR DELETE TO public USING (bucket_id = 'jugadores');

-- 3. Tabla de Partidos (incluye campo resultado)
CREATE TABLE IF NOT EXISTS public.partidos (
  id TEXT PRIMARY KEY,
  competicion TEXT NOT NULL,
  local TEXT NOT NULL,
  visitante TEXT NOT NULL,
  resultado TEXT DEFAULT '0 - 0',
  url_youtube TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Si la tabla partidos ya fue creada previamente, añade la columna resultado:
ALTER TABLE public.partidos ADD COLUMN IF NOT EXISTS resultado TEXT DEFAULT '0 - 0';

-- 3. Habilitar seguridad de nivel de fila (RLS)
ALTER TABLE public.jugadores ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.partidos ENABLE ROW LEVEL SECURITY;

-- 4. Permitir lectura, inserción, edición y borrado
CREATE POLICY "Permitir todo a jugadores" ON public.jugadores FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Permitir todo a partidos" ON public.partidos FOR ALL USING (true) WITH CHECK (true);
